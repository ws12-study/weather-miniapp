queryWeather: function (city) {
this.setData({ loading: true, error: '' });
 
if (this.data.mode === 'demo') {
this.renderXml(getDemoXml(city), 'demo');
return;
}
 
wx.request({
// ① 中文城市名必须 encodeURIComponent，否则 GET 查询串会乱码
url: WEATHER_WS_URL + '?theCityName=' + encodeURIComponent(city),
method: 'GET',
header: { 'content-type': 'text/xml; charset=utf-8' },
timeout: REQUEST_TIMEOUT,
 
// ② ★★★ 获取 XML 报文最关键的两行 ★★★
// dataType 默认 'json' 会强制 JSON.parse，XML 必然报错；
// 改成 'text'，res.data 就是服务端返回的原始 XML 字符串。
dataType: 'text', ← 关键：禁止框架 JSON.parse
responseType: 'text', ← 关键：按文本接收响应体
 
success: (res) => {
// 坑点：4xx / 5xx 在小程序里也会进 success，必须自己判状态码
if (res.statusCode !== 200 || typeof res.data !== 'string') {
this.fallbackToDemo(city, '接口返回 HTTP ' + res.statusCode);
return;
}
 
// ③ ★★★ res.data 此时就是完整的 XML 报文字符串 ★★★
const xmlText = res.data; ← 关键：拿到原始 XML 报文
console.log('[收到原始 XML 报文]\n' + xmlText);
 
try {
this.renderXml(xmlText, 'live');
} catch (err) {
this.fallbackToDemo(city, err.message);
}
},
 
fail: (err) => {
this.fallbackToDemo(city, (err && err.errMsg) || '网络请求失败');
}
});
},
 
renderXml: function (xmlText, source) {
// utils/xml-parser.js：清洗 → 实体解码 → 标签栈建树 → 抽取全部 <string>
const weather = parseWeatherXml(xmlText); ← 关键：XML → JS 对象
this.setData({ weather: weather, rawXml: xmlText, loading: false });
},
必圈重点 dataType: 'text' 与 responseType: 'text'：让 wx.request 不对 XML 做 JSON.parse； const xmlText = res.data：拿到完整 XML 报文； parseWeatherXml(xmlText)：手写解析器把 XML 转成 JS 对象。
